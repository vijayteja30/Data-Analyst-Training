Stored Procedures Syntax:
        Syntax1:
                CREATE OR REPLACE PROCEDURE procedure_name(parameter_list) 
                AS
                $$
                -- body of the stored procedure
                $$
                LANGUAGE SQL;
        Syntax2:
                CREATE OR REPLACE PROCEDURE procedure_name(parameter_list)
                LANGUAGE plpgsql AS
                BEGIN ATOMIC
                    -- body of the stored procedure
                END;
        

        # PostgreSQL Stored Procedure Example
        """
        CREATE OR REPLACE PROCEDURE log_user_activity(user_id INTEGER, activity TEXT)
        LANGUAGE plpgsql AS $$
        BEGIN
            INSERT INTO user_activity_log (user_id, activity, activity_time)
            VALUES (user_id, activity, NOW());
        END; $$;
        """
